function Translate(ch, l, linput, f) {
  l = l.substr(0,2);
  linput = linput.substr(0,2);
  // alert("traduction de "+ch+"\n\n"+linput+" --> "+l);
  var xhr = new XMLHttpRequest();
  xhr.open("POST", "../Translate.php?texte=" + ch + "&lang=" + l + "&langinput=" + linput + "&ROOMID=" + filename + "&lineid=" + currenttxt, true);

  xhr.onreadystatechange = function() { // Call a function when the state changes.
    if (this.readyState === XMLHttpRequest.DONE && this.status === 200) {
      // console.log(this.responseText);
      var response = JSON.parse(this.responseText);
      if ((! response["error"]) && response["data"]["translations"]) {
        Affiche(response["data"]["translations"][0]["translatedText"]);
        ParleThen(response["data"]["translations"][0]["translatedText"],f);
      } else {
        Affiche("---- Translation ERROR ----")
      }

    }
  };
  xhr.send();
}

function resizeslides() {
  var wx = window.innerWidth;
  var wy = window.innerHeight;
  var img = document.getElementById('slide');
  var imgmaxw = wx;
  var imgmaxh = (wy - 64 - 50 - 30);
  if (keepaspectratio) {
    if (imgmaxw/imgmaxh > aspectratio) {
      imgmaxw = imgmaxh*aspectratio;
    } else {
      imgmaxh = imgmaxw/aspectratio;
    }
  }
  img.style.width = imgmaxw+"px";
  img.style.height = imgmaxh+"px";
}
