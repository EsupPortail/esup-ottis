```php
<?php
/**
 * Validation des paramètres d'entrée
 */

/**
 * Valide un ROOMID
 * @param string $roomid L'ID de la salle à valider
 * @return bool True si valide, false sinon
 */
function validateRoomId($roomid) {
    if (!is_string($roomid)) {
        return false;
    }
    if (!preg_match('/^[a-zA-Z0-9_-]{1,64}$/', $roomid)) {
        return false;
    }
    return true;
}

/**
 * Valide un texte
 * @param string $text Le texte à valider
 * @param int $maxLength Longueur maximale autorisée
 * @return string Le texte sanitizé
 */
function sanitizeText($text, $maxLength = 10000) {
    if (!is_string($text)) {
        return '';
    }
    
    // Limiter la longueur
    if (strlen($text) > $maxLength) {
        $text = substr($text, 0, $maxLength);
    }
    
    // Supprimer les tags HTML
    $text = strip_tags($text);
    
    // Encoder les caractères spéciaux
    $text = htmlspecialchars($text, ENT_QUOTES | ENT_HTML5, 'UTF-8', false);
    
    // Normaliser les nouvelles lignes
    $text = str_replace(["\r\n", "\n", "\r"], ' ', $text);
    
    // Nettoyer les espaces multiples
    $text = preg_replace('/\s+/', ' ', $text);
    
    return trim($text);
}

/**
 * Valide un code de langue
 * @param string $lang Le code de langue à valider
 * @return string Le code de langue validé (ou 'fr' par défaut)
 */
function validateLanguage($lang) {
    if (!is_string($lang)) {
        return 'fr';
    }
    if (!preg_match('/^[a-zA-Z]{2,10}$/', $lang)) {
        return 'fr';
    }
    return $lang;
}

/**
 * Valide un lineid
 * @param mixed $lineid L'ID de ligne à valider
 * @return int L'ID validé
 */
function validateLineId($lineid) {
    if (!is_numeric($lineid)) {
        return 0;
    }
    return (int)$lineid;
}

/**
 * Vérifie qu'un fichier est dans le dossier tmp/ avec un préfixe valide
 * @param string $roomid L'ID de la salle
 * @param string $prefix Le préfixe du fichier
 * @return string Le chemin complet validé, ou false si invalide
 */
function validateTmpFilePath($roomid, $prefix = 'Link') {
    if (!validateRoomId($roomid)) {
        return false;
    }
    
    $allowedPrefixes = ['Link', 'Translated', 'Verylast', 'Image', 'ImageID', 'LinkQuestions', 'pincodes'];
    if (!in_array($prefix, $allowedPrefixes)) {
        return false;
    }
    
    return 'tmp/' . $prefix . '_' . $roomid;
}

/**
 * Ouvre un fichier tmp/ de manière sécurisée
 * @param string $roomid L'ID de la salle
 * @param string $prefix Le préfixe du fichier
 * @param string $mode Le mode d'ouverture
 * @return resource|bool Le pointeur de fichier, ou false en cas d'erreur
 */
function secureOpenTmpFile($roomid, $prefix, $mode) {
    $filepath = validateTmpFilePath($roomid, $prefix);
    if ($filepath === false) {
        return false;
    }
    
    // Vérifier que le fichier est bien dans le dossier tmp/
    if (strpos($filepath, 'tmp/') !== 0) {
        return false;
    }
    
    // Vérifier que le fichier n'a pas de path traversal
    if (strpos($filepath, '..') !== false || strpos($filepath, '/') !== 3) {
        return false;
    }
    
    $fd = @fopen($filepath, $mode);
    if ($fd === false) {
        error_log("Impossible d'ouvrir le fichier: " . $filepath);
        return false;
    }
    
    return $fd;
}
```

#### **Tâche 1.3: Implémenter la protection CSRF**
*ID: SEC-005 | Priorité: 🔴 | Durée: 1 heure | Complexité: Moyenne*

**Fichier à créer:** `/home/omist/Security/CSRF.php`

```php
<?php
/**
 * Protection CSRF pour l'application OMIST
 * 
 * Ce fichier fournit des fonctions pour générer et valider les tokens CSRF.
 * 
 * @package Security
 */

/**
 * Démarre la session si ce n'est pas déjà fait
 */
function ensureSessionStarted() {
    if (session_status() === PHP_SESSION_NONE) {
        session_start();
    }
}

/**
 * Génère un nouveau token CSRF
 * 
 * @return string Le token CSRF
 */
function generateCsrfToken() {
    ensureSessionStarted();
    
    if (empty($_SESSION['csrf_token'])) {
        $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
    }
    
    return $_SESSION['csrf_token'];
}

/**
 * Valide un token CSRF
 * 
 * @param string $token Le token à valider
 * @return bool True si le token est valide, false sinon
 */
function validateCsrfToken($token) {
    ensureSessionStarted();
    
    if (!isset($_SESSION['csrf_token'])) {
        return false;
    }
    
    // Utiliser hash_equals pour éviter les timing attacks
    return hash_equals($_SESSION['csrf_token'], $token);
}

/**
 * Régénère le token CSRF (à appeler après une validation réussie)
 */
function regenerateCsrfToken() {
    ensureSessionStarted();
    $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
}

/**
 * Vérifie si la requête est valide CSRF
 * 
 * @param string $method La méthode HTTP (POST, PUT, DELETE, etc.)
 * @return bool True si la requête est valide, false sinon
 */
function checkCsrfRequest($method = 'POST') {
    if ($_SERVER['REQUEST_METHOD'] !== $method) {
        return true; // Seules les requêtes POST/PUT/DELETE nécessitent CSRF
    }
    
    $token = $_POST['csrf_token'] ?? ($_SERVER['HTTP_X_CSRF_TOKEN'] ?? null);
    
    if (empty($token)) {
        error_log('CSRF: Token manquant');
        return false;
    }
    
    if (!validateCsrfToken($token)) {
        error_log('CSRF: Token invalide');
        return false;
    }
    
    // Régénérer le token après utilisation
    regenerateCsrfToken();
    
    return true;
}

/**
 * Affiche un champ caché CSRF pour les formulaires
 * 
 * @return string Le HTML du champ caché
 */
function csrfField() {
    return '<input type="hidden" name="csrf_token" value="' . htmlspecialchars(generateCsrfToken(), ENT_QUOTES, 'UTF-8') . '">';
}

/**
 * Retourne l'en-tête CSRF pour les requêtes AJAX
 * 
 * @return string L'en-tête HTTP
 */
function csrfMetaTag() {
    return '<meta name="csrf-token" content="' . htmlspecialchars(generateCsrfToken(), ENT_QUOTES, 'UTF-8') . '">';
}

/**
 * Middleware CSRF pour les scripts PHP
 * 
 * À inclure en haut des scripts qui traitent des requêtes POST/PUT/DELETE
 * 
 * @param string $method La méthode HTTP à vérifier (par défaut: POST)
 */
function requireCsrf($method = 'POST') {
    if (!checkCsrfRequest($method)) {
        http_response_code(403);
        header('Content-Type: application/json');
        echo json_encode([
            'error' => true,
            'message' => 'CSRF token invalid or missing'
        ]);
        exit;
    }
}

// Démarrer la session automatiquement
ensureSessionStarted();

// Régénérer le token à chaque nouvelle session
if (!isset($_SESSION['csrf_token_initiated'])) {
    $_SESSION['csrf_token_initiated'] = true;
    generateCsrfToken();
}
```

## **⚖️ CHOIX À FAIRE**

### **Choix 1: Méthode de validation du ROOMID**

**Options:**
1. **Regex simple:** `/^[a-zA-Z0-9_-]{1,64}$/` (recommandé)
   - Avantages: Simple, efficace
   - Inconvénients: Moins strict que nécessaire
   
2. **Regex stricte:** `/^[a-zA-Z0-9]{32}$/` (pour les MD5)
   - Avantages: Plus sécurisé, longueur fixe
   - Inconvénients: Moins flexible, nécessite de générer des ROOMID de 32 caractères

3. **Liste blanche:** Valider contre une liste de ROOMID existants
   - Avantages: Très sécurisé
   - Inconvénients: Complexe à maintenir

**Recommandation:** **Option 1** (Regex simple) - Équilibre entre sécurité et flexibilité

---

### **Choix 2: Implémentation du CSRF**

**Options:**
1. **Double Submit Cookie Pattern** (implémenté dans le code ci-dessus)
   - Avantages: Simple, compatible avec tous les navigateurs
   - Inconvénients: Nécessite des cookies

2. **Synchronizer Token Pattern** (alternative)
   - Avantages: Plus sécurisé
   - Inconvénients: Plus complexe à implémenter

3. **SameSite Cookie Attribute** (complémentaire)
   - Avantages: Protection supplémentaire
   - Inconvénients: Nécessite HTTPS, pas supporté par tous les navigateurs

**Recommandation:** **Option 1** (Double Submit Cookie) + **Option 3** (SameSite) pour une protection maximale

---

### **Choix 3: Gestion des erreurs en production**

**Options:**
1. **Désactiver complètement display_errors** (recommandé)
   - Avantages: Aucune fuite d'information
   - Inconvénients: Plus difficile à déboguer

2. **Afficher uniquement aux administrateurs**
   - Avantages: Facile à déboguer pour les admins
   - Inconvénients: Plus complexe à implémenter

3. **Page d'erreur personnalisée**
   - Avantages: Meilleure expérience utilisateur
   - Inconvénients: Nécessite plus de code

**Recommandation:** **Option 1** + page d'erreur 500 personnalisée

---

### **Choix 4: Stockage des fichiers temporaires**

**Options:**
1. **Fichiers plats dans tmp/** (actuel)
   - Avantages: Simple, pas de dépendance
   - Inconvénients: Pas scalable, pas de cleanup automatique

2. **Base de données** (MySQL, PostgreSQL)
   - Avantages: Plus sécurisé, scalable
   - Inconvénients: Nécessite une base de données, plus complexe

3. **Redis**
   - Avantages: Très rapide, TTL automatique
   - Inconvénients: Nécessite Redis, pas persistant

4. **SQLite**
   - Avantages: Simple, pas de serveur séparé, persistant
   - Inconvénients: Moins performant pour les requêtes simultanées

**Recommandation:** **Option 1** pour la Phase 1 (rapide à implémenter), puis migration vers **Option 2** ou **Option 4** pour les phases suivantes.

---

### **Choix 5: Gestion des tokens API**

**Options:**
1. **Fichier JSON externe** (actuel)
   - Avantages: Simple, facile à mettre à jour
   - Inconvénients: Sécurité limitée, pas de rotation automatique

2. **Variables d'environnement**
   - Avantages: Plus sécurisé, pas dans le code
   - Inconvénients: Plus difficile à gérer

3. **Service externe (Vault, AWS Secrets Manager)**
   - Avantages: Très sécurisé, rotation automatique
   - Inconvénients: Complexe, nécessite une infrastructure

**Recommandation:** **Option 1** pour maintenant, migration vers **Option 2** pour la production.
